hh hh hh*8 hh`).bank("ViscoSpaceDrum").slow(4).dec(.1).gain(slider(0.216))

// bassline (8 bars, with fills in the back half)
$: note(`<
[a1 ~ ~ [a1 a2] ~ g2 e2 ~]
[f1 ~ ~ f1 [~ c2] e2 ~ eb2]
[bb1 ~ ~ bb1 [~ d2] f2 ~ a1]
[e1 ~ ~ e2 ~ d2 g#1 [f1 g#1]]
[a1 ~ [~ a2] [a1 a2] ~ [g2 e2] ~ c2]
[f1 [~ f2] ~ f1 [c2 e2] ~ [f2 e2] eb2]
[bb1 ~ [bb2 bb1] ~ [~ d2] f2 [e2 d2] a1]
[e1 e1 e2 e1 [d2 d#2] e2 [g#1 g#2] [f1 g#1]]
>`)
  .s("sawtooth")
  .lpf(sine.range(300, 900).slow(8))
  .lpq(6)
  .attack(.005).decay(.2).sustain(.4).release(.1)
  .gain(slider(0.587)

// melody (call + answer)
$: note(`<
[e5@3 [~ e5] c5 d5 [e5 f5] g5]
[a5@3 g5 e5@2 [c5 b4] [~ c5]]
[d5@3 ~ f5 [e5 f5 e5] d5 a4]
[g#4@2 b4 d5 [f5 e5]@2 [d5 b4] [g#4 b4]]
[a5 [g5 e5] c6@2 [b5 a5 g5] e5 [~ a5] g5]
[[a5 c6] e6@2 [d6 c6 a5] g5 [e5 f5] [g5 a5] c6]
[d6@2 [c6 bb5] a5 [f5 d5] e5 [f5 a5 bb5] d6]
[[f6 e6 d6 b5] g#5 [f5 e5 d5 b4] g#4 [~ e5] [f5 g#5] [b5 d6] e6]
>`)
  .s("gm_voice_oohs")
  .off(1/8, x => x.add(note(12)).gain(.12))
  .lpf(2400)
  .attack(.01).decay(.3).sustain(.5).release(.3)
  .delay(.35).delaytime(.45).delayfeedback(.45)
  .room(.5)
.gain(slider(0))

// arpeggiator: follows the chords, filter opens over 16 bars
$: n("<[0 2 4 6 7 6 4 2]*2 [0 4 7 4 2 6 9 6]*2>")
  .scale("<A3:aeolian F3:lydian Bb3:lydian E3:mixolydian>")
  .s("sawtooth")
  .lpf(sine.range(400, 4000).slow(16))
  .lpq(8)
  .decay(.08).sustain(0)
  .pan(sine.slow(3))
  .sometimesBy(.15, x => x.ply(2))
  .gain(slider(0))
